"""Abstract base class for all market data feed adapters."""

from __future__ import annotations

from abc import ABC, abstractmethod

from algoforge.core.constants import Timeframe
from algoforge.core.models import OHLCV


class BaseFeed(ABC):
    """Abstract base for market data feed adapters.

    All feed adapters must implement these methods to be usable
    by the DataPipeline.
    """

    @abstractmethod
    async def connect(self) -> None:
        """Establish connection to the data source."""

    @abstractmethod
    async def disconnect(self) -> None:
        """Clean up connection resources."""

    @abstractmethod
    async def health_check(self) -> bool:
        """Verify the data source is reachable and working."""

    @abstractmethod
    async def fetch_historical(
        self,
        symbol: str,
        timeframe: Timeframe | None = None,
        period: str | None = None,
    ) -> list[OHLCV]:
        """Fetch historical OHLCV data."""

    @abstractmethod
    async def fetch_latest(
        self,
        symbol: str,
        timeframe: Timeframe | None = None,
    ) -> OHLCV | None:
        """Fetch the most recent candle."""
```

"""Factory for creating the appropriate data feed based on config."""

from algoforge.core.config import get_settings
from algoforge.data.feeds.base_feed import BaseFeed
from algoforge.data.feeds.yfinance_feed import YFinanceFeed
from algoforge.data.feeds.binance_feed import BinanceFeed
from algoforge.data.feeds.alphavantage_feed import AlphaVantageFeed


_FEED_MAP: dict[str, type[BaseFeed]] = {
    "yfinance": YFinanceFeed,
    "binance": BinanceFeed,
    "alphavantage": AlphaVantageFeed,
}


def create_feed(provider: str | None = None) -> BaseFeed:
    """Create the appropriate feed adapter.

    Args:
        provider: Feed provider name. If None, uses config.

    Returns:
        Configured feed adapter instance.

    Raises:
        ValueError: If provider is not supported.
    """
    if provider is None:
        provider = get_settings().data_feed.provider

    feed_cls = _FEED_MAP.get(provider)
    if feed_cls is None:
        raise ValueError(
            f"Unknown feed provider: {provider}. "
            f"Supported: {list(_FEED_MAP.keys())}"
        )
    return feed_cls()
```
